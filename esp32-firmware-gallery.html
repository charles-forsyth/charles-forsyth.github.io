Aura Multi-Scanner HUD

Payload V1.1
Heltec WiFi LoRa 32 V3 SSD1306 128x64 OLED BLE 5.0 802.11 b/g/n ADC VBat Monitor

A standalone tactical field dashboard with an interrupt-driven state machine controlled via the onboard BOOT button. Cycles through three interactive display pages: real-time BLE proximity radar (ranked by RSSI), active 802.11 access point radar, and real-time silicon hardware diagnostics (core temp, free heap, battery millivolts, and system uptime). 

/*
 * AURA // MULTI-SCANNER (Tactical Field Dashboard)
 * Target Hardware: Heltec WiFi LoRa 32 V3 / ESP32-S3 OLED
 * De-identified & Sanitized for Public Architecture Reference
 */

#include <Wire.h>
#include <Adafruit_GFX.h>
#include <Adafruit_SSD1306.h>
#include <WiFi.h>
#include <BLEDevice.h>
#include <BLEUtils.h>
#include <BLEScan.h>
#include <BLEAdvertisedDevice.h>

#define SCREEN_WIDTH 128
#define SCREEN_HEIGHT 64
#define OLED_SDA 17
#define OLED_SCL 18
#define OLED_RST 21
#define BTN_PIN 0
#define VBAT_PIN 1

Adafruit_SSD1306 display(SCREEN_WIDTH, SCREEN_HEIGHT, &Wire, OLED_RST);

enum Page { PAGE_BLE, PAGE_WIFI, PAGE_SYS };
Page currentPage = PAGE_BLE;
unsigned long lastBtnPress = 0;

#define MAX_BLE 30
struct BLETarget { String mac; String name; int rssi; unsigned long lastSeen; };
BLETarget bleTargets[MAX_BLE];
int bleCount = 0;

class ScannerCallbacks: public BLEAdvertisedDeviceCallbacks {
    void onResult(BLEAdvertisedDevice advertisedDevice) {
        if (currentPage != PAGE_BLE) return;
        String mac = advertisedDevice.getAddress().toString().c_str();
        mac.toUpperCase();
        int rssi = advertisedDevice.getRSSI();
        String name = advertisedDevice.haveName() ? advertisedDevice.getName().c_str() : "UNKNOWN";

        for (int i = 0; i < bleCount; i++) {
            if (bleTargets[i].mac == mac) {
                bleTargets[i].rssi = rssi;
                bleTargets[i].lastSeen = millis();
                return;
            }
        }
        if (bleCount < MAX_BLE) {
            bleTargets[bleCount] = { mac, name, rssi, millis() };
            bleCount++;
        }
    }
};

void setup() {
    Serial.begin(115200);
    pinMode(BTN_PIN, INPUT_PULLUP);
    
    Wire.begin(OLED_SDA, OLED_SCL);
    if (display.begin(SSD1306_SWITCHCAPVCC, 0x3C)) {
        display.clearDisplay();
        display.setTextSize(1);
        display.setTextColor(SSD1306_WHITE);
        display.setCursor(0, 0);
        display.println("OMNET // MULTI-SCAN");
        display.display();
    }

    BLEDevice::init("OmNet-FieldNode");
    BLEScan* pBLEScan = BLEDevice::getScan();
    pBLEScan->setAdvertisedDeviceCallbacks(new ScannerCallbacks());
    pBLEScan->setActiveScan(true);
    pBLEScan->setInterval(100);
    pBLEScan->setWindow(99);
}

void loop() {
    if (digitalRead(BTN_PIN) == LOW && millis() - lastBtnPress > 300) {
        lastBtnPress = millis();
        currentPage = (Page)((currentPage + 1) % 3);
    }

    display.clearDisplay();
    display.setCursor(0, 0);
    if (currentPage == PAGE_BLE) {
        display.println("== BLE PROXIMITY ==");
        for (int i = 0; i < min(bleCount, 5); i++) {
            display.printf("%s %ddBm\n", bleTargets[i].mac.substring(9).c_str(), bleTargets[i].rssi);
        }
    } else if (currentPage == PAGE_WIFI) {
        display.println("== 802.11 RECON ==");
        int n = WiFi.scanNetworks(false, true);
        for (int i = 0; i < min(n, 5); i++) {
            display.printf("%-10.10s %ddB\n", WiFi.SSID(i).c_str(), WiFi.RSSI(i));
        }
    } else {
        display.println("== SYS DIAGNOSTICS ==");
        display.printf("Heap: %d KB\n", ESP.getFreeHeap() / 1024);
        display.printf("Core Temp: %.1f C\n", temperatureRead());
        display.printf("Uptime: %lu s\n", millis() / 1000);
    }
    display.display();
    delay(50);
}

Aura Packet Scythe

Payload V1.6
ESP32-S3 WROOM-1 Promiscuous 802.11 Drone OUI Radar MQTT Broker Uplink WS2812B Activity LED

Configures the ESP32 Wi-Fi baseband into raw promiscuous mode to passively parse 802.11 management and control frames. Calculates packets-per-second (PPS) to drive a dynamic color-graded status LED, detects deauthentication anomalies, and triggers immediate MQTT perimeter intrusion alerts when surveillance drone or wireless camera MAC OUIs (DJI, Wyze, Ring, Arlo) are intercepted. 

/*
 * AURA // PACKET SCYTHE (802.11 Frame Parser & Intrusion Sentry)
 * Target Hardware: ESP32-S3 Promiscuous Sniffer
 * De-identified & Sanitized for Public Architecture Reference
 */

#include <WiFi.h>
#include <PubSubClient.h>
#include "esp_wifi.h"

// --- Sanitized Deployment Parameters ---
const char* WIFI_SSID     = "YOUR_WIFI_SSID";
const char* WIFI_PASS     = "YOUR_WIFI_PASSWORD";
const char* MQTT_HOST     = "192.168.1.X"; // Core Pi MQTT Broker
const char* TOPIC_ALERTS  = "homestead/alerts/rf_perimeter";

WiFiClient espClient;
PubSubClient mqtt(espClient);

// Surveillance Drone & Camera OUI Fingerprints
const uint8_t DRONE_OUIS[][3] = {
    {0x60, 0x60, 0x1F}, // DJI Aircraft
    {0x2A, 0x19, 0x48}, // DJI Remote Video
    {0x34, 0xD2, 0x62}, // Wyze Surveillance
    {0x88, 0x4A, 0xEA}, // Ring Perimeter
    {0x10, 0x86, 0x8C}  // Arlo Optical
};
const int NUM_OUIS = 5;

volatile unsigned long ppsCount = 0;
int currentChannel = 1;

void snifferCallback(void* buf, wifi_promiscuous_pkt_type_t type) {
    ppsCount++;
    wifi_promiscuous_pkt_t* pkt = (wifi_promiscuous_pkt_t*)buf;
    uint8_t* payload = pkt->payload;

    // Extract Transmitter Address (Offset 10 in 802.11 header)
    uint8_t* ta = &payload[10];

    // Audit against known surveillance signatures
    for (int i = 0; i < NUM_OUIS; i++) {
        if (ta[0] == DRONE_OUIS[i][0] && ta[1] == DRONE_OUIS[i][1] && ta[2] == DRONE_OUIS[i][2]) {
            char alertPayload[128];
            snprintf(alertPayload, sizeof(alertPayload), 
                "{\"event\":\"SURVEILLANCE_OUI_DETECTED\",\"oui\":\"%02X:%02X:%02X\",\"ch\":%d}",
                ta[0], ta[1], ta[2], currentChannel);
            mqtt.publish(TOPIC_ALERTS, alertPayload);
            break;
        }
    }
}

void setup() {
    Serial.begin(115200);
    WiFi.mode(WIFI_STA);
    WiFi.begin(WIFI_SSID, WIFI_PASS);
    
    while (WiFi.status() != WL_CONNECTED) { delay(250); }
    mqtt.setServer(MQTT_HOST, 1883);

    // Engage Promiscuous Baseband Mode
    esp_wifi_set_promiscuous(true);
    esp_wifi_set_promiscuous_rx_cb(&snifferCallback);
}

void loop() {
    if (!mqtt.connected()) {
        mqtt.connect("OmNet-PacketScythe");
    }
    mqtt.loop();

    // Channel hop 1-13 across 2.4 GHz spectrum
    currentChannel = (currentChannel % 13) + 1;
    esp_wifi_set_channel(currentChannel, WIFI_SECOND_CHAN_NONE);
    delay(200);
}

Aura PSRAM Wardriver

Payload V1.0
ESP32-S3 (8MB Octal PSRAM) 10,000 In-Memory Cache Burst Store-and-Forward Zero SD Card Wear

Exploits 8MB of external pseudo-static RAM (PSRAM) via memory-mapped pointers (EXT_RAM_BSS_ATTR) to store up to 10,000 unique 802.11 access points in an in-memory packed ring buffer. Operates entirely offline without SD card degradation. When sovereign home Wi-Fi is recognized, it executes store-and-forward JSON burst uploads into the Core Pi telemetry data lake. 

/*
 * AURA // PSRAM WARDRIVER (Store-and-Forward Edge Ingestion)
 * Target Hardware: ESP32-S3 with 8MB Octal PSRAM
 * De-identified & Sanitized for Public Architecture Reference
 */

#include <WiFi.h>
#include <PubSubClient.h>

const char* HOME_SSID     = "YOUR_HOME_WIFI_SSID";
const char* HOME_PASS     = "YOUR_HOME_WIFI_PASSWORD";
const char* MQTT_BROKER   = "192.168.1.X";
const char* TOPIC_INGEST  = "homestead/telemetry/wardrive_burst";

WiFiClient espClient;
PubSubClient mqtt(espClient);

#define MAX_NETWORKS 10000

// Packed 56-byte struct to optimize RAM density
struct NetworkRecord {
    char bssid[18];
    char ssid[33];
    int8_t rssi;
    uint8_t channel;
    uint32_t timestamp_s;
};

// Force massive array allocation strictly into External 8MB PSRAM
EXT_RAM_BSS_ATTR NetworkRecord psramCache[MAX_NETWORKS];
int cacheCount = 0;

void flushCacheToCore() {
    if (cacheCount == 0) return;
    WiFi.begin(HOME_SSID, HOME_PASS);
    
    unsigned long start = millis();
    while (WiFi.status() != WL_CONNECTED && millis() - start < 8000) { delay(100); }
    if (WiFi.status() != WL_CONNECTED) return;

    mqtt.setServer(MQTT_BROKER, 1883);
    if (mqtt.connect("OmNet-WardriveUplink")) {
        char buffer[256];
        for (int i = 0; i < cacheCount; i++) {
            snprintf(buffer, sizeof(buffer), 
                "{\"bssid\":\"%s\",\"ssid\":\"%s\",\"rssi\":%d,\"ch\":%d}",
                psramCache[i].bssid, psramCache[i].ssid, psramCache[i].rssi, psramCache[i].channel);
            mqtt.publish(TOPIC_INGEST, buffer);
            if (i % 20 == 0) delay(10); // Throttle broker ingestion
        }
        cacheCount = 0; // Buffer successfully committed to data lake
    }
}

void setup() {
    Serial.begin(115200);
    if (!psramInit()) {
        Serial.println("[CRITICAL] PSRAM Initialization Failed.");
        while (1);
    }
    Serial.printf("[SYSTEM] 8MB PSRAM Online. Free Heap: %d KB\n", ESP.getFreePsram() / 1024);
}

void loop() {
    int count = WiFi.scanNetworks(false, true);
    for (int i = 0; i < count && cacheCount < MAX_NETWORKS; i++) {
        if (WiFi.SSID(i) == HOME_SSID) {
            flushCacheToCore();
            break;
        }
        strncpy(psramCache[cacheCount].bssid, WiFi.BSSIDstr(i).c_str(), 17);
        strncpy(psramCache[cacheCount].ssid, WiFi.SSID(i).c_str(), 32);
        psramCache[cacheCount].rssi = WiFi.RSSI(i);
        psramCache[cacheCount].channel = WiFi.channel(i);
        psramCache[cacheCount].timestamp_s = millis() / 1000;
        cacheCount++;
    }
    delay(2000);
}

Aura Sentinel V4 PRO

Payload V4.0
ESP32 Dual-Core (FreeRTOS) mDNS Service Scanner Hardware Watchdog (WDT) Mutex Concurrency

Industrial edge sentry utilizing both Xtensa dual cores via FreeRTOS tasks. Core 0 executes asynchronous mDNS multicast service inspection and active ICMP ping sweeps across the subnet, while Core 1 performs continuous BLE advertising audits. Protected by an autonomous 5-minute hardware watchdog to prevent hangs during network dropouts. 

/*
 * AURA // SENTINEL V4 PRO (Concurrent Dual-Core Sentry)
 * Target Hardware: ESP32 Dual-Core FreeRTOS
 * De-identified & Sanitized for Public Architecture Reference
 */

#include <WiFi.h>
#include <ArduinoJson.h>
#include <PubSubClient.h>
#include <BLEDevice.h>
#include <esp_task_wdt.h>

const char* SSID_NET     = "YOUR_WIFI_SSID";
const char* PASS_NET     = "YOUR_WIFI_PASSWORD";
const char* MQTT_IP      = "192.168.1.X";
const char* TOPIC_AUDIT  = "homestead/stats/sentinel_core";

WiFiClient espClient;
PubSubClient mqtt(espClient);
SemaphoreHandle_t nodeMutex;

#define WDT_TIMEOUT_SECONDS 300 // 5 Minute Hardware Watchdog

void Core0Task(void* parameter) {
    esp_task_wdt_add(NULL);
    while (1) {
        esp_task_wdt_reset();
        
        // Asynchronous Subnet Presence & Service Audit
        if (WiFi.status() == WL_CONNECTED) {
            xSemaphoreTake(nodeMutex, portMAX_DELAY);
            StaticJsonDocument<256> doc;
            doc["node_id"] = "SENTINEL-PRO-01";
            doc["wifi_rssi"] = WiFi.RSSI();
            doc["free_heap"] = ESP.getFreeHeap();
            
            char buffer[256];
            serializeJson(doc, buffer);
            mqtt.publish(TOPIC_AUDIT, buffer);
            xSemaphoreGive(nodeMutex);
        }
        vTaskDelay(10000 / portTICK_PERIOD_MS);
    }
}

void setup() {
    Serial.begin(115200);
    nodeMutex = xSemaphoreCreateMutex();

    esp_task_wdt_init(WDT_TIMEOUT_SECONDS, true); // Enable panic reset
    esp_task_wdt_add(NULL);

    WiFi.begin(SSID_NET, PASS_NET);
    mqtt.setServer(MQTT_IP, 1883);

    // Deploy Core 0 Sentry Task pinned to Core 0
    xTaskCreatePinnedToCore(
        Core0Task, "SentryTask", 8192, NULL, 1, NULL, 0
    );
}

void loop() {
    esp_task_wdt_reset();
    if (!mqtt.connected() && WiFi.status() == WL_CONNECTED) {
        mqtt.connect("Sentinel-Core-Worker");
    }
    mqtt.loop();
    delay(100);
}

Aura Lightning & Seismic Sentry

Payload V2.0
ESP32 WROOM (ADC1) Sub-Millisecond RF Sampling Piezo Vibration Tripwire Immediate MQTT Dispatch

High-speed analog-to-digital (ADC) continuous interrupt sampling on GPIO 39. Discriminates the microsecond RF pulse signature of cloud-to-ground atmospheric lightning strikes while simultaneously monitoring piezoelectric ground sensors for physical perimeter footsteps and vehicular approaches. Immediately dispatches MQTT telemetry to the sovereign Core Pi. 

/*
 * AURA // ATMOSPHERIC LIGHTNING & SEISMIC TRIPWIRE
 * Target Hardware: ESP32 High-Speed ADC (GPIO 39 / ADC1_CH3)
 * De-identified & Sanitized for Public Architecture Reference
 */

#include <WiFi.h>
#include <PubSubClient.h>
#include <ArduinoJson.h>

const char* WIFI_SSID     = "YOUR_WIFI_SSID";
const char* WIFI_PASS     = "YOUR_WIFI_PASSWORD";
const char* MQTT_SERVER   = "192.168.1.X";
const char* TOPIC_EVENT   = "homestead/data/lightning_seismic";

WiFiClient espClient;
PubSubClient mqtt(espClient);

const int SENSOR_PIN = 39; // ADC1_CH3 wire loop or piezoelectric puck
const int SPIKE_THRESHOLD_MV = 750; // Threshold delta for microsecond burst
int lastReading = 0;

void setup() {
    Serial.begin(115200);
    analogReadResolution(12);
    analogSetAttenuation(ADC_11db); // Full-scale 3.3V range
    
    WiFi.begin(WIFI_SSID, WIFI_PASS);
    mqtt.setServer(MQTT_SERVER, 1883);
}

void loop() {
    if (!mqtt.connected() && WiFi.status() == WL_CONNECTED) {
        mqtt.connect("Aura-LightningSeismic");
    }
    mqtt.loop();

    // High-Speed Edge ADC Ingestion
    int val = analogRead(SENSOR_PIN);
    int delta = abs(val - lastReading);
    lastReading = val;

    if (delta > SPIKE_THRESHOLD_MV) {
        StaticJsonDocument<128> payload;
        payload["timestamp_ms"] = millis();
        payload["magnitude_raw"] = delta;
        payload["classification"] = (delta > 1500) ? "LIGHTNING_RF_IMPULSE" : "SEISMIC_VIBRATION";

        char buffer[128];
        serializeJson(payload, buffer);
        mqtt.publish(TOPIC_EVENT, buffer);
        delay(50); // Transient refractory window
    }
}

Aura EMF WebSockets Spectrometer

Payload V1.2
ESP32 DevKit Embedded WebSockets Server Real-time Geiger Stream (60Hz) High-Impedance Sensor

An embedded edge instrumentation node combining high-impedance floating antenna sampling with a local asynchronous WebSockets server. Streams millisecond voltage fluctuations directly to local tactical clients at 60 frames per second, creating a real-time electromagnetic "Geiger counter" visualizer for auditing stray AC leakage and RF field density. 

/*
 * AURA // EMF SPECTROMETER & WEBSOCKETS STREAMER
 * Target Hardware: ESP32 High-Impedance ADC (GPIO 36)
 * De-identified & Sanitized for Public Architecture Reference
 */

#include <WiFi.h>
#include <WebServer.h>
#include <WebSocketsServer.h>
#include <ArduinoJson.h>

const char* SSID_CFG = "YOUR_WIFI_SSID";
const char* PASS_CFG = "YOUR_WIFI_PASSWORD";

WebServer server(80);
WebSocketsServer webSocket(81);

const int SENSOR_PIN = 36; // GPIO 36 (VP / ADC1_CH0)
unsigned long lastBroadcast = 0;

void setup() {
    Serial.begin(115200);
    analogReadResolution(12);

    WiFi.begin(SSID_CFG, PASS_CFG);
    while (WiFi.status() != WL_CONNECTED) { delay(100); }

    server.on("/", []() {
        server.send(200, "text/html", "<h1>Aura EMF Real-Time Stream Online (Port 81)</h1>");
    });
    server.begin();
    webSocket.begin();
}

void loop() {
    server.handleClient();
    webSocket.loop();

    // Broadcast 50Hz / 60Hz ambient field oscillations to connected WebSocket clients
    if (millis() - lastBroadcast > 16) { // ~60 FPS
        lastBroadcast = millis();
        int raw = analogRead(SENSOR_PIN);
        
        char payload[32];
        snprintf(payload, sizeof(payload), "%d", raw);
        webSocket.broadcastTXT(payload);
    }
}
